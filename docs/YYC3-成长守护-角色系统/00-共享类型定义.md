## 一、使用说明

本文件为 `05-多元角色系统` 六篇代码片段补齐的**共享类型层**，用于让方法级节选具备类型上下文：

```typescript
// 在各角色文档的片段前引入
import type {
  Role, RoleInteraction, HealthRecord, Milestone,
  EmotionRecord, InterestSignal, CopingPattern,
} from "./00-共享类型定义";
```

- 角色文档中的代码为 `MultiRoleSystemManager` 的类内节选，**原文未附带类型定义与 import**；本文件按原文使用到的类型名补全声明。
- 标注为「结构占位」的类型仅给出可编译的结构契约，字段以原实现为准，落地时须按真实接口收敛。
- `GrowthLogAPI` 为声明式类，运行时由 API 层实现注入。

## 二、核心类型

```typescript
// ===== 角色与交互 =====
export interface Role {
  name: string;
  description: string;
  responsibilities: string[];
  capabilities: string[];
  [key: string]: any;
}

export interface RoleInteraction {
  description: string;
  type?: string;
  trigger?: string;
  actions?: string[];
  fromRole?: string;
  toRole?: string;
  createdAt?: number;
  [key: string]: any;
}

// ===== 成长记录与里程碑 =====
export interface HealthRecord {
  id: string;
  userId: string;
  type: string;
  date: Date | string;
  data: Record<string, unknown>;
  notes?: string;
  [key: string]: any;
}

export type CreateHealthRecordRequest = Omit<HealthRecord, "id">;

export interface Milestone {
  id?: string;
  userId?: string;
  category?: any;
  title?: string;
  description?: string;
  name?: string;
  expectedAge?: any;
  achievedDate?: any;
  evidence?: string[];
  [key: string]: any;
}

export type CreateMilestoneRequest = Omit<Milestone, "id"> & { userId: string; category: string; title: string };

export interface MilestoneMatrix {
  matrix: Record<string, Milestone[]>;
  statistics: MilestoneStatistics;
  generatedAt: Date | string;
  milestones?: Milestone[];
  byCategory?: Record<string, Milestone[]>;
  completed?: number;
  total?: number;
  [key: string]: any;
}

export interface MilestoneStatistics {
  totalMilestones: number;
  achievedMilestones: number;
  achievementRate: number;
  categoryStats: Record<string, any>;
  completed?: number;
  total?: number;
  completionRate?: number;
  byCategory?: Record<string, number>;
  [key: string]: any;
}

export interface DailyRhythm {
  userId: string;
  date: Date | string;
  sleep?: Record<string, unknown>;
  feeding?: Record<string, unknown>;
  activity?: Record<string, unknown>;
  mood?: string;
  [key: string]: any;
}

export type CreateDailyRhythmRequest = Omit<DailyRhythm, "userId">;

// ===== 情绪与兴趣 =====
export interface EmotionRecord {
  id: string;
  userId: string;
  emotion: string;
  intensity?: any;
  triggers?: string[];
  context?: string;
  recordedAt: Date | string;
  [key: string]: any;
}

export type EmotionalResponseRecord = EmotionRecord;
export type CreateEmotionalResponseRequest = Omit<EmotionRecord, "id">;

export interface InterestSignal {
  userId: string;
  signal?: string;
  context?: string;
  strength: number;
  category: string;
  persistence?: any;
  createdAt: Date | string;
  id?: string;
  duration?: number;
  observedAt?: Date | string;
  [key: string]: any;
}

export type InterestStage = "萌芽期" | "发展期" | "深化期" | "精通期" | string;
export type InterestCategory = string;
export type SkillLevel = any; // 原实现在不同角色中返回字符串或结构化对象
export type ScaffoldingLevel = string;
export type RitualType = string;
export type GuidanceApproach = any; // 原实现返回结构化对象

export interface CopingPattern {
  strategy: string;
  frequency: number;
  effectiveness: number;
  examples?: string[];
  [key: string]: any;
}

export interface ChallengeRecord {
  id: string;
  userId: string;
  challenge: string;
  response: string;
  outcome?: string;
  recordedAt?: Date | string;
  [key: string]: any;
}

// ===== 工具函数（原片段直接调用，未给出实现） =====
export const generateUUID = (): string =>
  (globalThis.crypto?.randomUUID?.() ?? Math.random().toString(36).slice(2));

// ===== API 客户端（声明式，运行时由 API 层注入实现） ======
export declare class GrowthLogAPI {
  createHealthRecord(request: CreateHealthRecordRequest): Promise<HealthRecord>;
  createMilestone(request: CreateMilestoneRequest): Promise<Milestone>;
  createEmotionalResponse(request: CreateEmotionalResponseRequest): Promise<EmotionalResponseRecord>;
  getMilestones(userId: string): Promise<Milestone[]>;
  getUser(userId: string): Promise<{ birthDate: any; [key: string]: any }>;
  [key: string]: any;
}
```

## 五、导出与约定

| 导出对象 | 类型 | 说明 |
| ---- | ---- | ---- |
| `Role` / `RoleInteraction` | interface | 角色注册与角色间交互契约 |
| `HealthRecord` / `Milestone` / `DailyRhythm` | interface | 记录者职责的数据载体 |
| `EmotionRecord` / `InterestSignal` / `CopingPattern` | interface | 聆听者、建议者职责的数据载体 |
| `InterestStage` / `SkillLevel` 等 | type | 标量枚举，统一为字符串联合 |
| `GrowthLogAPI` | declare class | 记录写入 API 的类型声明 |

维护约定：

1. 新增角色或职责涉及新类型时，先在此文件补声明，再在角色文档的「依赖声明」中引入。
2. 占位类型转正时，替换 `[key: string]: any` 为显式字段，并同步更新本文档变更历史。
3. 类型命名遵循 PascalCase，接口以名词结尾，联合类型以 `Stage` / `Level` / `Type` 结尾。
